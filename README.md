# tugasAkhirPBO
Tugas Akhir PBO yang diampu oleh Febriansyah Ramadhan, S.Kom., M.T.

Pada database hanya berisikan 1 table

create table public.siswa (
  id_siswa bigint generated by default as identity not null,
  nama character varying(100) not null,
  nisn character varying(30) not null,
  unit character varying(50) null,
  kelas character varying(50) null,
  alamat text null,
  nomor_telepon character varying(20) null,
  nilai_raport numeric(5, 2) null,
  created_at timestamp with time zone null default now(),
  constraint siswa_pkey primary key (id_siswa),
  constraint siswa_nisn_key unique (nisn)
) TABLESPACE pg_default;
